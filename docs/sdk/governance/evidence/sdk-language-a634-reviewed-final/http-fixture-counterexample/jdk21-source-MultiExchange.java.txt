477:     /** True if ALL ( even non-idempotent ) requests can be automatic retried. */
478:     private static final boolean RETRY_ALWAYS = retryPostValue();
479:     /** True if ConnectException should cause a retry. Enabled by default */
480:     static final boolean RETRY_CONNECT = !disableRetryConnect();

482:     /** Returns true is given request has an idempotent method. */
483:     private static boolean isIdempotentRequest(HttpRequest request) {
484:         String method = request.method();
485:         return switch (method) {
486:             case "GET", "HEAD" -> true;
487:             default -> false;
488:         };
489:     }
490: 
491:     /** Returns true if the given request can be automatically retried. */
492:     private static boolean canRetryRequest(HttpRequest request) {
493:         if (RETRY_ALWAYS)
494:             return true;
495:         if (isIdempotentRequest(request))
496:             return true;
497:         return false;
498:     }

500:     // Returns true if cancel(true) was called.
501:     // This is an important distinction in several scenarios:
502:     // for instance, if cancel(true) was called 1. we don't want
503:     // to retry, 2. we don't want to wrap the exception in
504:     // a timeout exception.
505:     boolean requestCancelled() {
506:         return interrupted.get() != null;
507:     }

509:     private boolean retryOnFailure(Throwable t) {
510:         if (requestCancelled()) return false;
511:         return t instanceof ConnectionExpiredException
512:                 || (RETRY_CONNECT && (t instanceof ConnectException));
513:     }

515:     private Throwable retryCause(Throwable t) {
516:         Throwable cause = t instanceof ConnectionExpiredException ? t.getCause() : t;
517:         return cause == null ? t : cause;
518:     }

520:     /**

522:      * completed exceptionally, or null.
523:      */
524:     private CompletableFuture<Response> getExceptionalCF(Throwable t) {
525:         if ((t instanceof CompletionException) || (t instanceof ExecutionException)) {
526:             if (t.getCause() != null) {
527:                 t = t.getCause();
528:             }
529:         }
530:         if (cancelled && !requestCancelled() && t instanceof IOException) {
531:             if (!(t instanceof HttpTimeoutException)) {
532:                 t = toTimeoutException((IOException)t);
533:             }
534:         } else if (retryOnFailure(t)) {
535:             Throwable cause = retryCause(t);
536: 
537:             if (!(t instanceof ConnectException)) {
538:                 // we may need to start a new connection, and if so
539:                 // we want to start with a fresh connect timeout again.
540:                 if (connectTimeout != null) connectTimeout.reset();
541:                 if (!canRetryRequest(currentreq)) {
542:                     return failedFuture(cause); // fails with original cause
543:                 }
544:             } // ConnectException: retry, but don't reset the connectTimeout.
545: 
546:             // allow the retry mechanism to do its work
547:             retryCause = cause;
548:             if (!expiredOnce) {
549:                 if (debug.on())
550:                     debug.log(t.getClass().getSimpleName() + " (async): retrying...", t);
551:                 expiredOnce = true;
552:                 // The connection was abruptly closed.
553:                 // We return null to retry the same request a second time.
554:                 // The request filters have already been applied to the
555:                 // currentreq, so we set previousreq = currentreq to
556:                 // prevent them from being applied again.
